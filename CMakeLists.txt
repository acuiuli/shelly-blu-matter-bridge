cmake_minimum_required(VERSION 3.20)
project(shelly_blu_bridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bridge_core core/src/bridge.cpp)
target_include_directories(bridge_core PUBLIC core/include)
target_compile_options(bridge_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(bridge_simulator simulator/main.cpp)
target_link_libraries(bridge_simulator PRIVATE bridge_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(bridge_tests tests/bridge_tests.cpp)
  target_link_libraries(bridge_tests PRIVATE bridge_core)
  target_compile_options(bridge_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME reconciliation COMMAND bridge_tests)
endif()
